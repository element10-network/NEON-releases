# Neon atom, first boot. NOTHING HERE NEEDS CHANGING for a RAK Hotspot V2 in Europe: put the
# card in, connect the network cable and start it. The atom names itself, installs itself
# (up to 30 minutes) and is then found at neon.directory, where its setup page asks for everything
# else: your name, the Meshtastic preset, the MeshCore channel (with a MeshCore board on USB),
# the panel password, the payout wallet, the position, and Wi-Fi. With no cable at all, join the
# atom's own Wi-Fi (Neon Setup ..., after 2 minutes) from a phone and choose the Wi-Fi on the page
# that opens: the atom joins it and installs.
# The outcome of the install is written to neon-status.txt next to this file.
#
# Only for other cases (KEY=VALUE, one per line; empty means the default):

# EU868 (default) or US915: the hotspot's band, fixed at install.
NEON_REGION=

# rak-v2 (default: RAK Hotspot V2) or nebra-rockpi4 (see hardware/devices/).
NEON_DEVICE=

# The Meshtastic preset to start with (default LongFast); the setup page asks for it anyway.
NEON_PRESET=

# Helium's gateway software with the hotspot's own key (default: on; its chip signs the atom's
# key, which photons need). 0 leaves it out.
NEON_HELIUM=

# SSH, to follow the install from another computer (empty: off, as shipped). A user name, and a
# password hash made with: openssl passwd -6 (removed once applied; without a user name nothing is
# applied), and/or a public key (one line: ssh-ed25519 AAAA... you@computer).
NEON_SSH_USER=
NEON_SSH_HASH=
NEON_SSH_KEY=

# Wi-Fi from the first boot, for an atom with no cable at all (otherwise join a network from the
# setup page later). The country is two letters (PL, DE, GB, US); the password 8 to 63 characters
# or the raw 64-hex-digit key. The password is removed from this file once installed. Country
# empty: PL in EU868, US in US915 (any EU country's 2.4 GHz Wi-Fi is the same).
NEON_WIFI_COUNTRY=
NEON_WIFI_SSID=
NEON_WIFI_PASSWORD=
